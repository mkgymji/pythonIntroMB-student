  \begin{tcolorbox}[
    colback=task-bg,
    colframe=microbit-blue,
    fonttitle=\bfseries,
    title={Úkol~\theukol\ifx\relax#1\relax\else{: #1}\fi},
    breakable,
    left=6pt, right=6pt, top=4pt, bottom=4pt
  ]%
}{%

}
