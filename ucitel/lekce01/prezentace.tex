\begin{frame}{Co je micro:bit?}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      Kapesní počítač navržený pro výuku programování.

      \bigskip
      \begin{itemize}
        \item \pojem{Displej:} 5 × 5 LED diod
        \item \pojem{Tlačítka:} A a B
        \item \pojem{Senzory:} pohyb, teplota, magnetismus
        \item \pojem{Zvuk:} reproduktor a mikrofon (v2)
        \item \pojem{Komunikace:} radio, Bluetooth, USB
      \end{itemize}

      \bigskip
      Programujeme ho v \textbf{Pythonu} —\\
      přesněji v jeho verzi zvané \textbf{MicroPython}.
    \end{column}
    \begin{column}{0.34\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        % tělo
        \draw[rounded corners=5pt, thick, fill=microbit-lightblue, draw=microbit-blue]

        \foreach \x in {0.32,0.80,1.28,1.76,2.24}

\begin{frame}{Vývojové prostředí: Thonny}

  \pojem{Thonny} je editor, ve kterém píšeme a spouštíme Python programy.

  \bigskip
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Horní okno — editor}
        Sem píšeme \textbf{program}.\\[4pt]
        Spuštění: klávesa \textbf{F5}\\
        nebo tlačítko \textbf{Run}.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Dolní okno — Shell}
        Tady vidíme výstupy\\
        a případné chybové hlášky.
      \end{block}
    \end{column}
  \end{columns}

  \bigskip
  \begin{block}{Nastavení pro micro:bit}
    \textbf{Nástroje → Nastavení → Interpreter}\\
    → vyber \textbf{MicroPython (BBC micro:bit)}
  \end{block}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Firmware: proč a jak}

  \pojem{Firmware} = software uvnitř micro:bitu.
  Bez správného firmware micro:bit Pythonu nerozumí.

  \bigskip
  Micro:bity \textbf{sdílíme mezi skupinami} — pokaždé zkontroluj.

  \medskip
  \begin{block}{Postup (stačí jednou na začátku hodiny)}
    \begin{enumerate}
      \item Připoj micro:bit přes USB
      \item Nastav interpreter (viz předchozí snímek)
      \item Klikni \textbf{Nainstalovat nebo aktualizovat MicroPython}
      \item Potvrď — instalace trvá asi 10 sekund
      \item V dolním panelu se objeví \texttt{MicroPython v...}
    \end{enumerate}
  \end{block}

  \smallskip
  \textcolor{gray}{\small Není to chyba. Je to normální součást sdíleného provozu.}
\end{frame}

\begin{frame}{Co je program?}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \pojem{Program} je seznam příkazů.\\
      Počítač je vykonává \textbf{odshora dolů}, jeden po druhém.

      \bigskip
      Počítač dělá přesně to, co napíšeme —\\
      nic víc, nic míň.

      \bigskip
      \textcolor{gray}{\small
        Soubor s Python programem\\
        se nazývá \textbf{skript} a má příponu \texttt{.py}.
      }
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}
        \foreach \i/\text in {0/příkaz 1, 1/příkaz 2, 2/příkaz 3}{

        \draw[->, thick, microbit-blue] (b1) -- (b2);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}[fragile]{Náš první program}

  \begin{lstlisting}
from microbit import *

display.scroll("Ahoj!")
  \end{lstlisting}

  \bigskip
  \begin{itemize}
    \item \pyinline{from microbit import *}\\
          načte všechny funkce pro práci s micro:bitem\\
          \textcolor{gray}{\small — tímto začíná každý náš program}
    \item \pyinline{display.scroll("Ahoj!")}\\
          skroluje text přes displej jako běžící nápis
  \end{itemize}

  \pause\bigskip
  \begin{alertblock}{Zkus}
    Nahraď \pyinline{scroll} za \pyinline{show}. Co se změní?
  \end{alertblock}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}[fragile]{\texttt{show} vs.\ \texttt{scroll}}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{display.show("A")}}
        Zobrazí \textbf{jeden znak}\\
        (nebo obrázek) staticky.\\[6pt]
        Rychlé, okamžité.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{display.scroll("Ahoj!")}}
        Text projíždí přes displej\\
        jako \textbf{běžící nápis}.\\[6pt]
        Pro delší texty.
      \end{block}
    \end{column}
  \end{columns}

  \bigskip
  Micro:bit má sadu připravených obrázků:

  \begin{lstlisting}
display.show(Image.HEART)
display.show(Image.HAPPY)
display.show(Image.PACMAN)
  \end{lstlisting}
\end{frame}

\begin{frame}{Co jsme se naučili}
  \begin{itemize}
    \item[\checkmark] Micro:bit — kapesní počítač s LED displejem, tlačítky, senzory
    \item[\checkmark] Thonny — editor pro psaní a spouštění programů
    \item[\checkmark] Firmware — software v micro:bitu; instalujeme z nastavení Thonny
    \item[\checkmark] Program = seznam příkazů, počítač je plní odshora dolů
    \item[\checkmark] Funkce = pojmenovaná akce, spouštíme ji závorkami
    \item[\checkmark] Argument = hodnota předaná funkci do závorek
    \item[\checkmark] \pyinline{from microbit import *} — začíná každý program
    \item[\checkmark] \pyinline{display.scroll()} — běžící nápis přes displej
    \item[\checkmark] \pyinline{display.show()} — zobrazí znak nebo obrázek
  \end{itemize}
\end{frame}
