\begin{frame}{Co je protokol}

  \begin{block}{Protokol}
    Přesně definovaná sada zpráv a pravidel — kdo posílá co a kdy.\\
    Bez protokolu si zařízení nerozumějí.
  \end{block}

  \bigskip
  \textbf{Příklad z~reálného světa:}
  \begin{itemize}
    \item HTTP — protokol pro webové stránky
    \item Bluetooth — protokol pro bezdrátová sluchátka
    \item \textbf{Náš protokol} — domluvíme si sami, tři zprávy
  \end{itemize}

  \bigskip
  \begin{exampleblock}{Dnešní protokol — reakcní hra}
    \begin{tabular}{ll}
      \pyinline{"NYNI"}    & hostitel → oba; signál k~reakci \\
      \pyinline{"HIT"}     & host → hostitel; „stiskl jsem A" \\
      \pyinline{"PROHRAL"} & hostitel → host; „byl jsi pomalejší" \\
    \end{tabular}
  \end{exampleblock}

\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Reakcní hra — pravidla}

  \begin{enumerate}
    \item \textbf{Hostitel} stiskne~B → čeká \emph{náhodnou} dobu (1–4~s)
    \item Hostitel pošle \pyinline{"NYNI"} → oba zobrazí \texttt{!}
    \item \textbf{Kdo dřív stiskne~A, vyhrál kolo:}
      \begin{itemize}
        \item Host stiskl dřív → pošle \pyinline{"HIT"}
        \item Hostitel stiskl dřív → nic neposílá
      \end{itemize}
    \item Hostitel vyhodnotí, informuje hosta, zobrazí skóre
  \end{enumerate}

  \bigskip
  \begin{center}
  \begin{tabular}{ccc}
    \toprule
    \textbf{Hostitel} & & \textbf{Host} \\
    \midrule
    stiskni B & & (čeká) \\
    sleep(náhoda) & & (čeká) \\
    send(\pyinline{"NYNI"}) & $\longrightarrow$ & zobraz \texttt{!} \\
    zobraz \texttt{!} & & \\
    \multicolumn{3}{c}{\textit{--- závodíme ---}} \\
    (nebo přijme) & $\longleftarrow$ & send(\pyinline{"HIT"}) \\
    send(\pyinline{"PROHRAL"}) & $\longrightarrow$ & (nebo přijme) \\
    \bottomrule
  \end{tabular}
  \end{center}

\end{frame}

from microbit import *
import radio
import random

radio.config(channel=42)
radio.on()

while True:

    \item \pyinline{random.randint(1000, 4000)} — náhoda ze~L05!

  \end{itemize}

\begin{frame}[fragile]{Program hosta}

\begin{lstlisting}[basicstyle=\tiny\ttfamily]
from microbit import *
import radio

radio.config(channel=42)
radio.on()
skore_ja = 0
skore_souper = 0

while True:
    display.show(Image.ASLEEP)
    prijato = radio.receive()

    if prijato == "NYNI":
        display.show("!")

        vitez = None
        while vitez is None:
            if button_a.was_pressed():
                vitez = "JA"
                radio.send("HIT")
            prijato2 = radio.receive()
            if prijato2 == "PROHRAL":
                vitez = "SOUPER"

        if vitez == "JA":
            skore_ja += 1
            display.show(Image.HAPPY)
        else:
            skore_souper += 1
            display.show(Image.SAD)

        sleep(1500)
        display.scroll(str(skore_ja) + ":" + str(skore_souper))
\end{lstlisting}

\begin{frame}{Asymetrický vs.\ symetrický protokol}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Reakcní hra — asymetrická}
        \begin{itemize}
          \item Hostitel a host mají \textbf{různý} kód
          \item Jedna strana „vede", druhá reaguje
          \item Jednodušší na implementaci
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{KNP — symetrická}
        \begin{itemize}
          \item Oba hráči spustí \textbf{stejný} kód
          \item Kdo „vede" určuje náhoda
          \item Složitější koordinace
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \bigskip
  \begin{exampleblock}{Otázka do třídy}
    Jak zajistíme, aby oba hráči věděli \emph{zároveň}, že kolo začíná —
    bez toho, abychom předem určili, kdo je hostitel?
  \end{exampleblock}

\end{frame}

\begin{frame}{KNP — synchronizace přes náhodný čas}

  \textbf{Každé zařízení} si losuje náhodnou dobu čekání (2–5~s).

  \bigskip
  \begin{center}
  \begin{tabular}{ccc}
    \toprule
    \textbf{Zařízení A} & & \textbf{Zařízení B} \\
    \midrule
    los: čekám 2,3~s & & los: čekám 3,8~s \\
    \textit{(čas vypršel — A bylo rychlejší)} & & \\
    zapípá + send(\pyinline{"GO"}) & $\longrightarrow$ & přijme \pyinline{"GO"} \\
    & & zapípá \\
    \multicolumn{3}{c}{\textit{--- oba zobrazí ? — výběr za 3 s ---}} \\
    send(\pyinline{"K"}) & $\longrightarrow$ & \\
    & $\longleftarrow$ & send(\pyinline{"N"}) \\
    \multicolumn{3}{c}{\textit{--- oba vyhodnotí ---}} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \bigskip
  \textbf{Výsledek:} Role „hostitele" vzniká dynamicky z~náhody —
  nikdo ji nemusí přidělovat předem.

\end{frame}

\begin{frame}[fragile]{KNP — synchronizace v kódu}

\begin{lstlisting}[basicstyle=\small\ttfamily]
cas_cekani = random.randint(2000, 5000)
konec = running_time() + cas_cekani

while running_time() < konec:
    zprava = radio.receive()
    if zprava == "GO":
        music.pitch(880, 200)  # soupeř byl rychlejší
        break
    sleep(50)
else:
    # while skončil přirozeně = čas mi vypršel = já jsem rychlejší
    music.pitch(880, 200)
    radio.send("GO")
\end{lstlisting}

  \begin{alertblock}{\texttt{while \ldots else}}
    \pyinline{else} u \pyinline{while} se spustí jen pokud smyčka skončila
    \emph{přirozeně} (podmínka nepravdivá) — \textbf{ne} přes \pyinline{break}.\\
    Tady: „přirozený konec" = čas vypršel = jsem rychlejší.
  \end{alertblock}

\end{frame}
